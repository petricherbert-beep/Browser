async function init() { // Protobuf initialisieren const root = protobuf.parse(protoSource).root; DeviceListProto = root.lookupType("DeviceListProto"); // Abruf mit Protobuf Accept-Header const res = await api("/devices", { headers: { 'Accept': 'application/x-protobuf' } }); if (res.status === 401) { alert("Passwort falsch!"); return; } document.getElementById("login-overlay").style.display = "none"; document.getElementById("status").innerText = "● Verbunden"; document.getElementById("status").style.color = "var(--accent)"; document.getElementById("srv-url").innerText = BASE.replace("https://", ""); socket = io(BASE, { extraHeaders: { "x-api-key": API_KEY } }); socket.on("location_update", d => { console.log("🚀 Socket Update:", d); // Debug-Log if (d.accident && (!devices[d.deviceId] || !devices[d.deviceId].accident)) { map.setView([d.lat, d.lon], 18); followId = d.deviceId; } updateDevice(d); }); socket.on("geofences_updated", list => drawGF(list)); // 🔥 BINÄRDATEN DEKODIEREN + CAMELCASE FIX const buffer = await res.arrayBuffer(); if (buffer.byteLength > 0) { const decoded = DeviceListProto.decode(new Uint8Array(buffer)); // longs: Number -> Verhindert Zeitstempel-Fehler // camelcase: true -> Macht aus device_id automatisch deviceId const data = DeviceListProto.toObject(decoded, { longs: Number, camelcase: true }); console.log("📦 Geladene Geräte:", data.devices); // Debug-Log const coords = []; (data.devices || []).forEach(d => { updateDevice(d); if (d.lat && d.lon) coords.push([d.lat, d.lon]); }); if (coords.length > 0) { map.fitBounds(coords, { padding: [50, 50], maxZoom: 16 }); } } refreshGF(); }
